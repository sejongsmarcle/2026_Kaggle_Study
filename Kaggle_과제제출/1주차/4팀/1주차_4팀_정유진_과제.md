### 4. 전처리 + 특징값 생성

```python
# 4. 전처리 + 특징값 생성

# train/test 합치기(같은 방식으로 전처리하려고)
all_df = pd.concat([train, test], sort=False).reset_index(drop=True)

# 객실 등급별 평균 요금으로 Fare 결측치 채우기
fare_mean = all_df.groupby("Pclass")["Fare"].transform("mean")
all_df["Fare"] = all_df["Fare"].fillna(fare_mean)

# 이름에서 호칭 추출
all_df["honorific"] = all_df["Name"].str.extract(r",\s*([^.]*)\.")[0].str.strip()

# 호칭별 평균 나이로 Age 결측치 채우기
age_mean = all_df.groupby("honorific")["Age"].transform("mean")
all_df["Age"] = all_df["Age"].fillna(age_mean)

# 함께 탄 가족 수 추가
all_df["family_num"] = all_df["SibSp"] + all_df["Parch"]

# 혼자 탔는지 여부 추가
all_df["alone"] = (all_df["family_num"] == 0).astype(int)

# 사용하지 않을 열 제거
all_df = all_df.drop(["PassengerId", "Name", "Ticket", "Cabin"], axis=1)

# 탑승 항구 결측치 처리
all_df["Embarked"] = all_df["Embarked"].fillna("missing")

# 문자형 데이터를 숫자로 변환
from sklearn.preprocessing import LabelEncoder

for col in ["Sex", "Embarked", "honorific"]:
    le = LabelEncoder()
    all_df[col] = le.fit_transform(all_df[col])

# 전처리 후 결측치 확인
all_df.isnull().sum()
```
