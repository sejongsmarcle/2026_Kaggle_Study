# ------------------------------------------------------------
# 1. 모델 학습시키기

# LightGBM 라이브러리를 불러온다.
# LightGBM은 여러 개의 결정 트리를 이용해 예측하는 머신러닝 모델이다.
import lightgbm as lgb

# 현재 설치된 LightGBM 버전을 확인한다.
print(lgb.__version__)


# Train 데이터와 Validation 데이터를 나누기 위한 함수를 불러온다.
from sklearn.model_selection import train_test_split

# 나중에 K-Fold 교차검증을 사용하기 위한 라이브러리도 불러온다.
from sklearn.model_selection import KFold


# 전체 학습 데이터를 Train과 Validation으로 나눈다.
# train_X : 생존 예측에 사용할 입력 데이터
# train_Y : 실제 생존 여부(0 또는 1)
# test_size=0.2 : 전체 데이터의 20%를 Validation 데이터로 사용
X_train, X_valid, y_train, y_valid = train_test_split(
    train_X,
    train_Y,
    test_size=0.2
)


# LightGBM에게 범주형 변수라는 것을 알려줄 변수 목록이다.
# 숫자로 변환되어 있어도 실제로는 범주형 데이터인 변수들이다.
categories = ["Embarked", "Pclass", "Sex", "honorific", "alone"]


# Train 데이터를 LightGBM이 학습할 수 있는 형태로 만든다.
lgb_train = lgb.Dataset(
    X_train,
    y_train,
    categorical_feature=categories
)


# Validation 데이터를 LightGBM이 평가할 수 있는 형태로 만든다.
lgb_eval = lgb.Dataset(
    X_valid,
    y_valid,
    categorical_feature=categories,

    # Train 데이터의 설정을 참고하도록 한다.
    reference=lgb_train
)


# LightGBM의 기본 설정값을 지정한다.
lgbm_params = {
    # 생존/사망처럼 결과가 0 또는 1인 이진 분류 문제를 사용한다.
    "objective": "binary",

    # 랜덤 요소가 있을 때 같은 결과를 재현하기 위한 seed 값이다.
    "random_seed": 1234,
}


# ------------------------------------------------------------
# 실제로 모델을 학습한다.
# ------------------------------------------------------------

model_lgb = lgb.train(
    lgbm_params,       # 위에서 설정한 모델 옵션
    lgb_train,         # 학습에 사용할 Train 데이터
    valid_sets=lgb_eval,  # 학습 중 Validation 데이터로 성능 확인

    # 최대 100개의 트리를 순서대로 학습한다.
    num_boost_round=100,

    # Validation 성능이 5번 동안 좋아지지 않으면 학습을 중단한다.
    # → 과적합을 방지하기 위한 Early Stopping
    early_stopping_rounds=5,

    # 10번 학습할 때마다 학습 상황을 출력한다.
    verbose_eval=10
)


# ------------------------------------------------------------
# 2. 성능 확인하기
# ------------------------------------------------------------

# Validation 데이터의 생존 확률을 예측한다.
# 결과는 0 또는 1이 아니라 0~1 사이의 확률로 나온다.
y_pred = model_lgb.predict(
    X_valid,

    # Early Stopping이 선택한 가장 좋은 학습 횟수를 사용한다.
    num_iteration=model_lgb.best_iteration
)


# 예측값과 실제 정답의 정확도를 계산하기 위한 함수를 불러온다.
from sklearn.metrics import accuracy_score


# 확률값을 0 또는 1로 바꾼 후 실제 정답과 비교한다.
# 예: 0.73 → 1 / 0.24 → 0
accuracy_score(

    np.round(y_pred)
)


# ------------------------------------------------------------
# 3. 파라미터 값을 바꿔보며 성능 올려보기
# ------------------------------------------------------------

# LightGBM의 하이퍼파라미터를 변경한다.
# → 과제에서 "파라미터 값을 바꿔보며 성능 올려보기"에 해당한다.
lgbm_params = {
    # 이진 분류 문제이므로 그대로 binary를 사용한다.
    "objective": "binary",

    # 하나의 트리에서 사용할 데이터 구간의 개수를 조절한다.
    "max_bin": 331,

    # 하나의 트리에서 만들 수 있는 최대 leaf 수를 설정한다.
    "num_leaves": 20,

    # leaf가 되기 위해 필요한 최소 데이터 수를 설정한다.
    "min_data_in_leaf": 57,

    # 동일한 결과를 재현하기 위한 seed 값이다.
    "random_seed": 1234,
}


# 변경한 파라미터로 사용할 Train 데이터를 다시 만든다.
lgb_train = lgb.Dataset(
    X_train,
    y_train,
    categorical_feature=categories
)


# 변경한 파라미터로 사용할 Validation 데이터를 다시 만든다.
lgb_eval = lgb.Dataset(
    X_valid,
    y_valid,
    categorical_feature=categories,
    reference=lgb_train
)


# 변경된 하이퍼파라미터를 이용해 모델을 다시 학습한다.
model_lgb = lgb.train(
    lgbm_params,
    lgb_train,
    valid_sets=lgb_eval,

    # 최대 100번까지 학습한다.
    num_boost_round=100,

    # 이번에는 성능이 20번 연속 좋아지지 않으면 중단한다.
    early_stopping_rounds=20,

    verbose_eval=10
)


# 새로 만든 모델로 Validation 데이터의 생존 확률을 예측한다.
y_pred = model_lgb.predict(
    X_valid,
    num_iteration=model_lgb.best_iteration
)


# 변경 전 모델과 변경 후 모델의 성능을 비교한다.
accuracy_score(
    y_valid,
    np.round(y_pred)
)

# 4. 가장 성능 좋은 모델로 최종 마무리
# ------------------------------------------------------------

# K-Fold 교차검증에서 데이터를 3개 그룹으로 나눈다.
folds = 3

kf = KFold(n_splits=folds)


# 각 Fold에서 만들어진 모델을 저장할 리스트를 만든다.
models = []


# 데이터를 3번 나누어 학습/검증을 반복한다.
for train_index, val_index in kf.split(train_X):

    # 현재 Fold의 Train 데이터를 가져온다.
    X_train = train_X.iloc[train_index]

    # 현재 Fold의 Validation 데이터를 가져온다.
    X_valid = train_X.iloc[val_index]

    # 현재 Fold의 Train 정답 데이터를 가져온다.
    y_train = train_Y.iloc[train_index]

    # 현재 Fold의 Validation 정답 데이터를 가져온다.
    y_valid = train_Y.iloc[val_index]


    # 현재 Fold의 Train 데이터를 LightGBM 형식으로 만든다.
    lgb_train = lgb.Dataset(
        X_train,
        y_train,
        categorical_feature=categories
    )


    # 현재 Fold의 Validation 데이터를 만든다.
    lgb_eval = lgb.Dataset(
        X_valid,
        y_valid,
        categorical_feature=categories,
        reference=lgb_train
    )


    # 현재 Fold의 데이터를 이용해 모델을 학습한다.
    model_lgb = lgb.train(
        lgbm_params,
        lgb_train,
        valid_sets=lgb_eval,
        num_boost_round=100,
        early_stopping_rounds=20,
        verbose_eval=10,
    )


    # Validation 데이터의 생존 확률을 예측한다.
    y_pred = model_lgb.predict(
        X_valid,
        num_iteration=model_lgb.best_iteration
    )


    # 현재 Fold에서의 정확도를 확인한다.
    print(
        accuracy_score(
            y_valid,
            np.round(y_pred)
        )
    )


    # 학습한 모델을 리스트에 저장한다.
    models.append(model_lgb)
