```python
import numpy as np
import pandas as pd
import warnings
import random
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import LabelEncoder
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score
import lightgbm as lgb

warnings.filterwarnings("ignore")

# 각 데이터셋 csv 파일 불러오기
train_df = pd.read_csv("./data/train.csv")               
test_df = pd.read_csv("./data/test.csv")                 
submission = pd.read_csv("./data/gender_submission.csv") 

# 랜덤 시드 설정
np.random.seed(1234)
random.seed(1234)     

# 출력 옵션 설정
pd.set_option("display.max_columns", 50)
pd.set_option("display.max_rows", 50)    

plt.style.use('ggplot')

# --------------------------------------------------------
# 1. 임의의 열 추출 
# 2. dropna()를 이용하여 결측치 제거 
# 3. groupby()를 이용하여 Embarked와 Survived로만 집계 
# 4. unstack()를 이용하여 데이터를 수평으로 변환 
embarked_df = train_df[["Embarked","Survived","PassengerId"]].dropna().groupby(["Embarked","Survived"]).count().unstack()

embarked_df.plot.bar(stacked=True) # 누적 막대 그래프로 표현
plt.show() # 그래프 출력 

# .iloc[]를 이용하여 전체 승객 대비 비율 계산, survived_rate열에 할당 
embarked_df["survived_rate"] = embarked_df.iloc[:,0]/(embarked_df.iloc[:,0] + embarked_df.iloc[:,1])

# 성별 및 객실 등급별 생존자 수 누적 막대 그래프 생성 
sex_df = train_df[["Sex","Survived","PassengerId"]].dropna().groupby(["Sex","Survived"]).count().unstack()
sex_df.plot.bar(stacked=True)
plt.show()

ticket_df = train_df[["Pclass","Survived","PassengerId"]].dropna().groupby(["Pclass","Survived"]).count().unstack()
ticket_df.plot.bar(stacked=True)
plt.show()

# 생존 여부에 따른 연령 분포 시각화 
plt.hist(x=[train_df.Age[train_df.Survived==0], train_df.Age[train_df.Survived==1]], bins=8, histtype='barstacked', label=["Death","Survived"])
plt.legend()
plt.show()

# 카테고리 변수인 탑승 항구와 성별을 0과 1로 더미화 하기 
train_df_corr = pd.get_dummies(train_df, columns=["Sex"], drop_first=True)
train_df_corr = pd.get_dummies(train_df_corr, columns=["Embarked"])

# train_corr 변수에 train_df_corr 데이터의 상관관계 행렬 저장 (숫자형 열만 지정)
train_corr = train_df_corr.corr(numeric_only=True)

# 상관 행렬 그래프 
plt.figure(figsize=(9, 9))
sns.heatmap(train_corr, vmax=1, vmin=-1, center=0, annot=True)
plt.show()

# 학습 데이터셋과 테스트 데이터셋 통합 
all_df = pd.concat([train_df, test_df], sort=False).reset_index(drop=True)

# 결측치 메우기 (Pclass 기준 평균 Fare)
Fare_mean = all_df[["Pclass","Fare"]].groupby("Pclass").mean().reset_index()
Fare_mean.columns = ["Pclass","Fare_mean"]

all_df = pd.merge(all_df, Fare_mean, on="Pclass", how="left")
all_df.loc[(all_df["Fare"].isnull()), "Fare"] = all_df["Fare_mean"]
all_df = all_df.drop("Fare_mean", axis=1)

# name 열을 기준으로 성, 호칭, 이름으로 분리 (파라미터 n, regex 추가)
name_df = all_df["Name"].str.split("[,.]", n=2, expand=True, regex=True)
name_df.columns = ["family_name","honorific","name"]

# 문자열 공백 제거, 기존 데이터에 새로운 열로 병합 
name_df["family_name"] = name_df["family_name"].str.strip()
name_df["honorific"] = name_df["honorific"].str.strip()
name_df["name"] = name_df["name"].str.strip()
all_df = pd.concat([all_df, name_df], axis=1)

# 호칭별 연령 분포 박스플롯 및 평균 연령 확인 
plt.figure(figsize=(18, 5))
sns.boxplot(x="honorific", y="Age", data=all_df)
plt.show()

# 분리한 이름 데이터와 병합 
train_df = pd.concat([train_df, name_df[0:len(train_df)].reset_index(drop=True)], axis=1)
test_df = pd.concat([test_df, name_df[len(train_df):].reset_index(drop=True)], axis=1)

# 호칭별 생존율 시각화 
honorific_df = train_df[["honorific","Survived","PassengerId"]].dropna().groupby(["honorific","Survived"]).count().unstack()
honorific_df.plot.bar(stacked=True)
plt.show()

# age열 결측치 메우기 
honorific_age_mean = all_df[["honorific","Age"]].groupby("honorific").mean().reset_index()
honorific_age_mean.columns = ["honorific","honorific_Age"]

all_df = pd.merge(all_df, honorific_age_mean, on="honorific", how="left")
all_df.loc[(all_df["Age"].isnull()), "Age"] = all_df["honorific_Age"]
all_df = all_df.drop(["honorific_Age"], axis=1)

# 부모/자식과 형제/배우자 수 합 
all_df["family_num"] = all_df["Parch"] + all_df["SibSp"]

# 가족 수가 0이라면 alone에 1 할당 
all_df.loc[all_df["family_num"] == 0, "alone"] = 1
all_df["alone"] = all_df["alone"].fillna(0)

# 불필요한 변수 삭제 
all_df = all_df.drop(["PassengerId", "Name", "family_name", "name", "Ticket", "Cabin"], axis=1)

# 데이터 타입이 문자열인 변수 목록 추출 
categories = all_df.columns[all_df.dtypes == "object"]

# 주요 호칭외 나머지 호칭 통합 
all_df.loc[~((all_df["honorific"] == "Mr") |
    (all_df["honorific"] == "Miss") |
    (all_df["honorific"] == "Mrs") |
    (all_df["honorific"] == "Master")), "honorific"] = "other"

# LabelEncoder를 이용하여 남은 문자열 범주형 데이터를 숫자로 변환 
all_df["Embarked"] = all_df["Embarked"].fillna("missing")

for cat in categories:
    le = LabelEncoder()
    if all_df[cat].dtypes == "object":    
        all_df[cat] = le.fit_transform(all_df[cat])

# 통합한 데이터를 다시 분리 
train_X = all_df[~all_df["Survived"].isnull()].drop("Survived", axis=1).reset_index(drop=True)
train_Y = train_df["Survived"]
test_X = all_df[all_df["Survived"].isnull()].drop("Survived", axis=1).reset_index(drop=True)

# --------------------------------------------------------
# 1. 모델 학습시키기

# 홀드 아웃 기법을 사용하여 학습 데이터와 검증 데이터로 8:2 분리
X_train, X_valid, y_train, y_valid = train_test_split(train_X, train_Y, test_size=0.2, random_state=1234)

# 카테고리 변수 리스트 지정
cat_features = ["Embarked", "Pclass", "Sex", "honorific", "alone"]

# LightGBM용 데이터셋 작성
lgb_train = lgb.Dataset(X_train, y_train, categorical_feature=cat_features)
lgb_eval = lgb.Dataset(X_valid, y_valid, categorical_feature=cat_features, reference=lgb_train)

# 머신 러닝 모델 하이퍼파라미터 초기 설정
lgbm_params = {
    "objective": "binary",        
    "random_seed": 1234,
}

# 모델 훈련 진행
model_lgb = lgb.train(lgbm_params, 
                      lgb_train, 
                      valid_sets=[lgb_train, lgb_eval], 
                      num_boost_round=100,
                      callbacks=[lgb.early_stopping(stopping_rounds=5), lgb.log_evaluation(period=10)])

# 2. 성능 확인하기
# 각 변수의 중요도 시각화
importance = pd.DataFrame(model_lgb.feature_importance(), index=X_train.columns, columns=["importance"]).sort_values(by="importance", ascending=True)
importance.plot.barh()
plt.show()

# 검증 세트를 사용하여 예측 수행 및 정확도 평가
y_pred = model_lgb.predict(X_valid, num_iteration=model_lgb.best_iteration)
print("초기 모델 정확도:", accuracy_score(y_valid, np.round(y_pred)))

# 3. 파라미터 값 바꿔보며 성능 올려보기
lgbm_params_tuned = {
    "objective": "binary",
    "max_bin": 331,
    "num_leaves": 20, #트리의 최대 잎 개수. 모델 복잡도를 결정. 값이 클 수록 정확도는 높아지나 과적합 위험
    "min_data_in_leaf": 57, #잎의 최소 데이터 수. 과적합 방지
    "random_seed": 1234,
}

lgb_train = lgb.Dataset(X_train, y_train, categorical_feature=cat_features)
lgb_eval = lgb.Dataset(X_valid, y_valid, categorical_feature=cat_features, reference=lgb_train)

# 4. 가장 성능 좋은 모델로 최종 마무리
model_lgb_tuned = lgb.train(lgbm_params_tuned, 
                            lgb_train, 
                            valid_sets=[lgb_train, lgb_eval], 
                            num_boost_round=100,
                            callbacks=[lgb.early_stopping(stopping_rounds=20), lgb.log_evaluation(period=10)])

# 최종 튜닝된 모델의 예측 결과 산출 및 정확도 재확인
y_pred_tuned = model_lgb_tuned.predict(X_valid, num_iteration=model_lgb_tuned.best_iteration)
print("튜닝 모델 정확도:", accuracy_score(y_valid, np.round(y_pred_tuned)))

# 테스트 데이터 예측 수행 및 CSV 파일로 결과 출력
test_pred = model_lgb_tuned.predict(test_X, num_iteration=model_lgb_tuned.best_iteration)
submission["Survived"] = np.round(test_pred).astype(int)
submission.to_csv("./data/final_submission.csv", index=False)
```
